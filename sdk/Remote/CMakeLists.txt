cmake_minimum_required(VERSION 3.16)
project(PtglRemoteSender VERSION 0.1.0 LANGUAGES CXX)
get_filename_component(PTGL_REMOTE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(PtglRemoteSender INTERFACE)
add_library(ptgl::remoteSender ALIAS PtglRemoteSender)
set_target_properties(PtglRemoteSender PROPERTIES EXPORT_NAME remoteSender)
target_compile_features(PtglRemoteSender INTERFACE cxx_std_17)
if(MSVC)
    target_compile_options(PtglRemoteSender INTERFACE /EHsc /utf-8)
endif()
target_include_directories(PtglRemoteSender INTERFACE
    $<BUILD_INTERFACE:${PTGL_REMOTE_ROOT}>
    $<INSTALL_INTERFACE:include>)
target_include_directories(PtglRemoteSender SYSTEM INTERFACE
    $<BUILD_INTERFACE:${PTGL_REMOTE_ROOT}/thirdparty/Remote/websocketpp-0.8.2>
    $<BUILD_INTERFACE:${PTGL_REMOTE_ROOT}/thirdparty/Remote/asio-asio-1-28-0/asio/include>)
if(WIN32)
    target_link_libraries(PtglRemoteSender INTERFACE ws2_32 mswsock)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(PtglRemoteSender INTERFACE Threads::Threads)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(PtglRemoteSenderConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/PtglRemoteSenderConfig.cmake"
    INSTALL_DESTINATION lib/cmake/PtglRemoteSender)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/PtglRemoteSenderConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS PtglRemoteSender EXPORT PtglRemoteSenderTargets)
install(EXPORT PtglRemoteSenderTargets NAMESPACE ptgl:: DESTINATION lib/cmake/PtglRemoteSender)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/PtglRemoteSenderConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/PtglRemoteSenderConfigVersion.cmake" DESTINATION lib/cmake/PtglRemoteSender)
install(DIRECTORY "${PTGL_REMOTE_ROOT}/ptgl/Remote" DESTINATION include/ptgl)
install(DIRECTORY "${PTGL_REMOTE_ROOT}/thirdparty/Remote/websocketpp-0.8.2/websocketpp" DESTINATION include)
install(DIRECTORY "${PTGL_REMOTE_ROOT}/thirdparty/Remote/asio-asio-1-28-0/asio/include/" DESTINATION include
    FILES_MATCHING PATTERN "*.hpp" PATTERN "*.ipp")
install(FILES "${PTGL_REMOTE_ROOT}/thirdparty/Remote/websocketpp-0.8.2/COPYING" DESTINATION share/licenses/PtglRemoteSender/WebSocketpp)
install(FILES "${PTGL_REMOTE_ROOT}/thirdparty/Remote/asio-asio-1-28-0/asio/LICENSE_1_0.txt" DESTINATION share/licenses/PtglRemoteSender/Asio)
install(FILES "${PTGL_REMOTE_ROOT}/thirdparty/Remote/Readme.md" "${PTGL_REMOTE_ROOT}/LICENSE" DESTINATION share/licenses/PtglRemoteSender)
option(PTGL_REMOTE_SENDER_TESTS "Build standalone sender tests without rendering dependencies" OFF)
option(PTGL_REMOTE_SENDER_DEMO "Build the standalone sender example" OFF)
if(PTGL_REMOTE_SENDER_TESTS)
    enable_testing()
    add_executable(RemoteProtocolTests "${PTGL_REMOTE_ROOT}/tests/Remote/ProtocolTests.cpp")
    target_link_libraries(RemoteProtocolTests PRIVATE ptgl::remoteSender)
    add_test(NAME RemoteProtocol COMMAND RemoteProtocolTests)
    add_executable(RemoteSenderTests "${PTGL_REMOTE_ROOT}/tests/Remote/SenderTests.cpp" "${PTGL_REMOTE_ROOT}/tests/Remote/SecondTranslationUnit.cpp")
    target_link_libraries(RemoteSenderTests PRIVATE ptgl::remoteSender)
    add_test(NAME RemoteSender COMMAND RemoteSenderTests)
    set_tests_properties(RemoteSender PROPERTIES TIMEOUT 20)
endif()
if(PTGL_REMOTE_SENDER_DEMO)
    add_executable(RemotePublisherDemo "${PTGL_REMOTE_ROOT}/examples/RemoteDemo/Publisher.cpp")
    target_link_libraries(RemotePublisherDemo PRIVATE ptgl::remoteSender)
endif()
